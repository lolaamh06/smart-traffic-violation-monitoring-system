MySQL Schema Definition (CREATE TABLE DDL) ▾ 
-- 1. VEHICLES Table
CREATE TABLE vehicles (
  vehicle_id VARCHAR(36) PRIMARY KEY,
  reg_number VARCHAR(15) UNIQUE NOT NULL,
  owner_name VARCHAR(100) NOT NULL,
  owner_phone VARCHAR(15) NOT NULL,
  owner_email VARCHAR(100),
  vehicle_type ENUM('Car', 'Bike', 'Truck', 'Auto', 'Bus') NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. OFFICERS Table
CREATE TABLE officers (
  officer_id VARCHAR(36) PRIMARY KEY,
  badge_number VARCHAR(15) UNIQUE NOT NULL,
  name VARCHAR(100) NOT NULL,
  zone VARCHAR(50) NOT NULL,
  username VARCHAR(50) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. LOCATIONS Table
CREATE TABLE locations (
  location_id VARCHAR(36) PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  latitude DECIMAL(9,6) NOT NULL,
  longitude DECIMAL(9,6) NOT NULL,
  zone VARCHAR(50) NOT NULL
);

-- 4. VIOLATION_TYPES Table
CREATE TABLE violation_types (
  type_id VARCHAR(36) PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  base_fine INT NOT NULL
);

-- 5. VIOLATIONS Table (Fact Table with 4 FKs)
CREATE TABLE violations (
  violation_id VARCHAR(36) PRIMARY KEY,
  vehicle_id VARCHAR(36) NOT NULL,
  officer_id VARCHAR(36) NOT NULL,
  location_id VARCHAR(36) NOT NULL,
  type_id VARCHAR(36) NOT NULL,
  violation_time TIMESTAMP NOT NULL,
  evidence_url TEXT,
  status ENUM('Pending', 'Contested', 'Closed') DEFAULT 'Pending',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (vehicle_id) REFERENCES vehicles(vehicle_id) ON DELETE CASCADE,
  FOREIGN KEY (officer_id) REFERENCES officers(officer_id),
  FOREIGN KEY (location_id) REFERENCES locations(location_id),
  FOREIGN KEY (type_id) REFERENCES violation_types(type_id)
);

-- 6. FINES Table (1:1 with Violations)
CREATE TABLE fines (
  fine_id VARCHAR(36) PRIMARY KEY,
  violation_id VARCHAR(36) UNIQUE NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  due_date TIMESTAMP NOT NULL,
  paid_at TIMESTAMP NULL,
  payment_method ENUM('Cash', 'Online', 'UPI'),
  is_paid BOOLEAN DEFAULT FALSE,
  FOREIGN KEY (violation_id) REFERENCES violations(violation_id) ON DELETE CASCADE
);

-- 7. ROUTE_SEARCHES Table (Junction / Query log)
CREATE TABLE route_searches (
  search_id VARCHAR(36) PRIMARY KEY,
  user_id VARCHAR(100) NOT NULL,
  start_point VARCHAR(200) NOT NULL,
  end_point VARCHAR(200) NOT NULL,
  searched_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Query #8: Correlated Subquery SUBQUERY
Purpose: Select violations where the generated fine is higher than the average fine of its violation category.
MySQL SQL 
SELECT v.violation_id, f.amount
FROM violations v
JOIN fines f ON v.violation_id = f.violation_id
WHERE f.amount > (
  SELECT AVG(f2.amount)
  FROM violations v2
  JOIN fines f2 ON v2.violation_id = f2.violation_id
  WHERE v2.type_id = v.type_id
);
Firebase Firestore JS 
// Client-side double map to aggregate averages per type
const allFines = (await getDocs(collection(db, 'fines'))).docs.map(d => d.data());
const allVios = (await getDocs(collection(db, 'violations'))).docs.map(d => ({id: d.id, ...d.data()}));

// Calculate avg by type
const typeFines = {};
allVios.forEach(v => {
  const fine = allFines.find(f => f.violationId === v.id);
  if (fine) {
    if (!typeFines[v.typeId]) typeFines[v.typeId] = [];
    typeFines[v.typeId].push(fine.amount);
  }
});

const averages = {};
Object.keys(typeFines).forEach(tid => {
  const sum = typeFines[tid].reduce((a,b) => a+b, 0);
  averages[tid] = sum / typeFines[tid].length;
});

const results = allVios.filter(v => {
  const fine = allFines.find(f => f.violationId === v.id);
  return fine && fine.amount > averages[v.typeId];
});

Query #16: TRIGGER Definition TRIGGER
Purpose: Automatically create fine invoice and calculate 1.5x escalation if repeat offender (>3 violations, 90 days).
MySQL SQL 
CREATE TRIGGER after_violation_insert
AFTER INSERT ON violations FOR EACH ROW
BEGIN
  DECLARE baseVal INT;
  DECLARE isRepeat BOOLEAN;
  
  SELECT base_fine INTO baseVal 
  FROM violation_types WHERE type_id = NEW.type_id;

  SET isRepeat = (
    SELECT COUNT(*) > 3 
    FROM violations 
    WHERE vehicle_id = NEW.vehicle_id 
      AND violation_time >= DATE_SUB(NEW.violation_time, INTERVAL 90 DAY)
  );

  INSERT INTO fines (fine_id, violation_id, amount, due_date, is_paid)
  VALUES (
    UUID(), 
    NEW.violation_id, 
    IF(isRepeat, baseVal * 1.5, baseVal), 
    DATE_ADD(NEW.violation_time, INTERVAL 30 DAY), 
    FALSE
  );
END;
Firebase Firestore JS 
// Firestore Cloud Function Trigger in functions/index.js
exports.onViolationCreated = functions.firestore
  .document('violations/{violationId}')
  .onCreate(async (snap, context) => {
    const data = snap.data();
    const typeRef = await db.collection('violation_types').doc(data.typeId).get();
    const baseFine = typeRef.data().baseFine;

    // Check repeat offences
    const ninetyDaysAgo = new Date(data.violationTime.toDate() - 90 * 86400000);
    const prevCountSnap = await db.collection('violations')
      .where('vehicleId', '==', data.vehicleId)
      .where('violationTime', '>=', ninetyDaysAgo)
      .get();

    const isRepeat = prevCountSnap.size > 3;
    const finalAmount = isRepeat ? baseFine * 1.5 : baseFine;

    return db.collection('fines').add({
      violationId: context.params.violationId,
      amount: finalAmount,
      dueDate: new Date(Date.now() + 30 * 86400000),
      isPaid: false,
      paidAt: null,
      paymentMethod: null
    });
  });
